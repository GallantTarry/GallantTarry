(function () {
  var stage = document.getElementById('tk-hw-stage');
  var tip = document.getElementById('tk-hw-tooltip');
  if (!stage) return;
  function makeFallbackSvg(label) {
    var base = (label || '?').split(' · ')[0].trim();
    var parts = base.split(/\s+/);
    var abbr = parts.length >= 2
      ? (parts[0][0] + parts[parts.length - 1][0]).toUpperCase()
      : base.slice(0, 2).toUpperCase();
    var svg = '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 48 48">' +
      '<rect width="48" height="48" rx="8" fill="#18181b"/>' +
      '<text x="24" y="29" text-anchor="middle" font-family="sans-serif" font-size="16" font-weight="700" fill="#10b981">' +
      abbr.replace(/[<>&]/g, '') + '</text></svg>';
    return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg);
  }
  function applyFallback(img) {
    if (img.dataset.fallbackApplied) return;
    img.dataset.fallbackApplied = '1';
    img.src = makeFallbackSvg(img.getAttribute('alt') || img.dataset.label);
  }
  stage.querySelectorAll('img.tk-hw-tile').forEach(function (img) {
    if (img.complete && img.naturalWidth === 0) applyFallback(img);
    img.addEventListener('error', function () { applyFallback(img); });
  });
  if (!tip) return;
  function showTip(img) {
    var rect = img.getBoundingClientRect();
    tip.textContent = img.dataset.label || img.alt || '';
    tip.style.left = Math.min(Math.max(rect.left + rect.width / 2, 70), window.innerWidth - 70) + 'px';
    tip.style.top = (rect.top - 22) + 'px';
    tip.style.opacity = '1';
  }
  stage.addEventListener('mouseover', function (e) {
    var img = e.target.closest('img.tk-hw-tile');
    if (img) showTip(img);
  });
  stage.addEventListener('mouseout', function (e) {
    if (e.target.closest('img.tk-hw-tile')) tip.style.opacity = '0';
  });
  stage.addEventListener('click', function (e) {
    var img = e.target.closest('img.tk-hw-tile');
    if (img) showTip(img);
    else tip.style.opacity = '0';
  });
})();
</script>
